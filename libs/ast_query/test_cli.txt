# Tests run_cli_str against fixtures for exact formatted-string output.
# fixture line numbers: helper=5, greet=9, Greeter=13, __init__=14, speak=17

# --- def subcommand ---

call: ast_query.run_cli_str
arg cmdline = "def helper libs/ast_query/fixtures/sample.py"

# --- pattern subcommand (JS, full mode only — requires ast-grep) ---
# Two hits: function helper at line 5, function greet at line 9.
# ast-grep reports 0-based col; search_pattern adds +1, so col=1 for col-0 functions.
# text = full matched source including body (newlines embedded in text field).

call: ast_query.run_cli_str
arg cmdline = "pattern js \"function $NAME($$$) { $$$ }\" libs/ast_query/fixtures/sample.js"
normalize: function helper\(x\) \{[\s\S]*?\} => <helper-body>
normalize: function greet\(name\) \{[\s\S]*?\} => <greet-body>
expect return = "libs/ast_query/fixtures/sample.js:5:1\t<helper-body>\nlibs/ast_query/fixtures/sample.js:9:1\t<greet-body>"

# --- git subcommand ---

# rev-parse --is-inside-work-tree returns "true" in a git work tree.
call: ast_query.run_cli_str
arg cmdline = "git rev-parse --is-inside-work-tree"
expect return = "true"

# Two flags: both outputs on separate lines.
call: ast_query.run_cli_str
arg cmdline = "git rev-parse --is-inside-work-tree --is-bare-repository"
expect return = "true\nfalse"

# Global -H 1 composes with git: limits to the first line.
call: ast_query.run_cli_str
arg cmdline = "-H 1 git rev-parse --is-inside-work-tree --is-bare-repository"
expect return = "true"

# Mutating subcommand is rejected.
call: ast_query.run_cli_str
arg cmdline = "git commit -m x"
expect return = "git: 'commit' is not an allowed read-only subcommand"

# -C token is rejected (leading dash, not in allowlist).
call: ast_query.run_cli_str
arg cmdline = "git -C /tmp status"
expect return = "git: '-C' is not an allowed read-only subcommand"

# `branch` is excluded deliberately, not an oversight: the allowlist is
# subcommand-granular, so admitting `branch` would also admit
# `git branch -d`/`-m`. `git worktree` is excluded for the same reason
# (`worktree add`/`remove`/`prune`). See design/look.md §9.5.
call: ast_query.run_cli_str
arg cmdline = "git branch"
expect return = "git: 'branch' is not an allowed read-only subcommand"

# Missing subcommand returns the missing-subcommand message.
call: ast_query.run_cli_str
arg cmdline = "git"
expect return = "git: missing subcommand"

# git composes inside seq: echo then git, joined by newline.
call: ast_query.run_cli_str
arg cmdline = "seq 'echo === STATE ===' 'git rev-parse --is-inside-work-tree'"
expect return = "=== STATE ===\ntrue"

# -H before git inside a seq string: inner command trimmed to 1 line, then TAIL appended.
# rev-parse returns "true\nfalse"; -H 1 trims to "true"; seq appends echo output.
call: ast_query.run_cli_str
arg cmdline = "seq '-H 1 git rev-parse --is-inside-work-tree --is-bare-repository' 'echo TAIL'"
expect return = "true\nTAIL"

# --- global --log / --log-repeat flags ---

# --log warning (a recognised, no-op-here value): the data channel is
# unchanged from the un-flagged grep TODO result — nothing logs a warning
# for this command, so no second section appears.
call: ast_query.run_cli_str
arg cmdline = "--log warning grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "libs/ast_query/fixtures/grep_sample.md:3:1\tTODO finish this."

# --log info: same reasoning — nothing logs at info yet, so raising the
# threshold must not disturb the data channel.
call: ast_query.run_cli_str
arg cmdline = "--log info grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "libs/ast_query/fixtures/grep_sample.md:3:1\tTODO finish this."

# --log-repeat each: with an empty log, the repeat mode has nothing to
# collapse or expand, so the data channel is unchanged.
call: ast_query.run_cli_str
arg cmdline = "--log-repeat each grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "libs/ast_query/fixtures/grep_sample.md:3:1\tTODO finish this."

# Composing -H with --log, in either order: proves the two globals are
# peeled independently and neither disturbs the other's effect on data.
call: ast_query.run_cli_str
arg cmdline = "-H 1 --log info grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "libs/ast_query/fixtures/grep_sample.md:3:1\tTODO finish this."

# An unrecognised --log value logs an error instead of being applied; ctx
# keeps its default threshold, so the error itself always renders, and the
# data channel is suppressed entirely (design/look.md §9.1.2).
call: ast_query.run_cli_str
arg cmdline = "--log bogus grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "**ERROR** --log: unknown value 'bogus' (expected one of: error, warning, info) **ERROR**"

# Same story for an unrecognised --log-repeat value.
call: ast_query.run_cli_str
arg cmdline = "--log-repeat bogus grep TODO libs/ast_query/fixtures/grep_sample.md"
expect return = "**ERROR** --log-repeat: unknown value 'bogus' (expected one of: collapse, each) **ERROR**"

# --- --exclude on the searching subcommands (def/refs/outline/pattern/grep) ---

# def -r control: helper is found in fixtures/sample.py without --exclude.
call: ast_query.run_cli_str
arg cmdline = "def -r helper libs/ast_query"
expect contains = "libs/ast_query/fixtures/sample.py:5:0\tfunction helper"

# def -r --exclude fixtures: the whole fixtures/ subtree is pruned from the
# walk (never descended into), so the sample.py hit disappears entirely.
call: ast_query.run_cli_str
arg cmdline = "def -r helper libs/ast_query --exclude fixtures"
expect return = ""

# An explicitly-named excluded path is rejected too (like GNU grep), even
# though outline's single PATH operand isn't a directory walk.
call: ast_query.run_cli_str
arg cmdline = "outline libs/ast_query/fixtures/sample.py --exclude sample.py"
expect return = ""

# grep -r --exclude "*.md": a full-path glob prunes grep_sample.md; the three
# grep_sample.py hits (lines 8/10/12) and the extensionless noext_sample hit
# remain, since neither matches *.md.
call: ast_query.run_cli_str
arg cmdline = "grep -r TODO libs/ast_query/fixtures --exclude \"*.md\""
expect return = "libs/ast_query/fixtures/grep_sample.py:8:3\t# TODO in a comment\nlibs/ast_query/fixtures/grep_sample.py:10:15\tmsg = \"string TODO here\"\nlibs/ast_query/fixtures/grep_sample.py:12:1\tTODO_count = 0\nlibs/ast_query/fixtures/noext_sample:2:3\t# TODO extensionless fixture"

# comma-separated form: --exclude a,b excludes both grep_sample.md and
# noext_sample in one flag, leaving only the grep_sample.py hits.
call: ast_query.run_cli_str
arg cmdline = "grep -r TODO libs/ast_query/fixtures --exclude grep_sample.md,noext_sample"
expect return = "libs/ast_query/fixtures/grep_sample.py:8:3\t# TODO in a comment\nlibs/ast_query/fixtures/grep_sample.py:10:15\tmsg = \"string TODO here\"\nlibs/ast_query/fixtures/grep_sample.py:12:1\tTODO_count = 0"

# repeated-flag form is equivalent to the comma form directly above.
call: ast_query.run_cli_str
arg cmdline = "grep -r TODO libs/ast_query/fixtures --exclude grep_sample.md --exclude noext_sample"
expect return = "libs/ast_query/fixtures/grep_sample.py:8:3\t# TODO in a comment\nlibs/ast_query/fixtures/grep_sample.py:10:15\tmsg = \"string TODO here\"\nlibs/ast_query/fixtures/grep_sample.py:12:1\tTODO_count = 0"

# trailing slash is tolerated: --exclude fixtures/ behaves exactly like
# --exclude fixtures (compare to the def -r --exclude fixtures stanza above).
call: ast_query.run_cli_str
arg cmdline = "def -r helper libs/ast_query --exclude fixtures/"
expect return = ""

# refs -r --exclude fixtures: the fixtures/ subtree is pruned, so the
# helper reference in sample.py disappears entirely.
call: ast_query.run_cli_str

# --- --exclude on the walking source subcommands (ls/find/each) ---

# (whose name doesn't match the bare "fixtures" component) is untouched.
call: ast_query.run_cli_str
arg cmdline = "find libs/ast_query --ext .py --exclude fixtures"

# ls --exclude on an explicitly-named path is rejected too (like GNU grep),
# even though a plain path arg isn't itself a directory walk.
call: ast_query.run_cli_str
arg cmdline = "ls libs/ast_query/fixtures --exclude fixtures"
expect return = ""

# --exclude written BEFORE the positional ROOT (every other --exclude stanza
# in this file puts the flag last, to dodge LG-14): fixtures/ is pruned from
# the walk of libs/ast_query itself, so the listing still contains the
# top-level ast_query.py. This is the assertion that closed LG-13; avoid
# pinning an exact full listing here since that directory also contains
# volatile entries such as __pycache__.
call: ast_query.run_cli_str
arg cmdline = "ls --exclude fixtures libs/ast_query"
expect contains = "ast_query.py"

# ls -r control: no --exclude, fixtures/ contributes its own listing plus its
# five files (gap_fixtures/ and __pycache__/ excluded here only to keep the
# expected output stable; that is not what this stanza is testing).
call: ast_query.run_cli_str
arg cmdline = "ls -r libs/ast_query --exclude gap_fixtures,__pycache__"
expect return = "libs/ast_query/README.md\nlibs/ast_query/__init__.py\nlibs/ast_query/__main__.py\nlibs/ast_query/ast_query.py\nlibs/ast_query/fixtures/\nlibs/ast_query/test_cli.txt\nlibs/ast_query/test_dsl.txt\nlibs/ast_query/test_gaps.txt\nlibs/ast_query/test_jsbackend.txt\nlibs/ast_query/test_primary.txt\nlibs/ast_query/fixtures/grep_sample.md\nlibs/ast_query/fixtures/grep_sample.py\nlibs/ast_query/fixtures/noext_sample\nlibs/ast_query/fixtures/sample.js\nlibs/ast_query/fixtures/sample.py"

# ls -r with fixtures added to the exclude set: the fixtures/ directory
# itself is pruned from the walk, so neither it nor any of its five files
# appear, unlike the control stanza directly above.
call: ast_query.run_cli_str
arg cmdline = "ls -r libs/ast_query --exclude gap_fixtures,__pycache__,fixtures"

# comma-separated form: --exclude a,b excludes both grep_sample.md and
# noext_sample in one flag, leaving only the .js/.py fixture files.
call: ast_query.run_cli_str
arg cmdline = "ls libs/ast_query/fixtures --exclude grep_sample.md,noext_sample"

# repeated-flag form is equivalent to the comma form directly above.

# trailing slash is tolerated: --exclude fixtures/ behaves exactly like

call: ast_query.run_cli_str

# --- check: syntax check (LG-10) ---

# A clean file checks silently: no data, no log.
call: ast_query.run_cli_str
arg cmdline = "check libs/ast_query/fixtures/sample.py"
expect return = ""

# A file with a syntax error renders **ERROR** wrapping
# "path:line:col: SyntaxError: msg" (exact message text varies by CPython
# version, so only the wrapper + path:line: prefix are pinned).
call: ast_query.run_cli_str
arg cmdline = "check libs/ast_query/gap_fixtures/broken_syntax.py"
expect contains = "**ERROR**"
expect contains = "libs/ast_query/gap_fixtures/broken_syntax.py:7:"

# Several paths in one invocation: the clean ones stay silent, only the
# broken one is reported.
call: ast_query.run_cli_str
arg cmdline = "check libs/ast_query/fixtures/sample.py libs/ast_query/gap_fixtures/broken_syntax.py libs/ast_query/fixtures/grep_sample.py"
expect contains = "**ERROR**"
expect contains = "libs/ast_query/gap_fixtures/broken_syntax.py:7:"

# A non-.py path draws a plain (unwrapped) warning, not an error — v1 is
# Python-only.
call: ast_query.run_cli_str
arg cmdline = "check libs/ast_query/fixtures/grep_sample.md"
expect return = "libs/ast_query/fixtures/grep_sample.md: not checked — v1 only syntax-checks Python (.py) files"

# check is a DSL collector: piped file paths become operands, so the
# broken file under gap_fixtures/ is reported (const_sample.py in the
# same directory is clean and stays silent). Keep the optional --ext
# after the positional ROOT per LG-14.
call: ast_query.run_cli_str
arg cmdline = "find libs/ast_query/gap_fixtures --ext .py ++ check"
expect contains = "**ERROR**"
expect contains = "libs/ast_query/gap_fixtures/broken_syntax.py:7:"
