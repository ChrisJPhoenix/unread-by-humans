# --- LG-15: argparse errors print the least useful message (top-level ---
# --- usage, no flags) at the moment the surface is needed.            ---
# NOT MECHANICALLY PINNABLE today: run_cli_str/run_cli_argv call
# parser.parse_args(), and argparse's error() path writes usage text to
# sys.stderr (via parser.error() -> print_usage(sys.stderr) -> exit),
# not to sys.stdout. libs/test_harness/harness.py only wraps the call in
# `redirect_stdout` (harness.py:24,279,283) to satisfy `expect prints`;
# it never redirects stderr, and `expect raises`/`expect return` cannot
# see written-but-not-returned text either. So there is currently no
# harness field that observes the usage string, only the SystemExit
# itself (already covered by the LG-3/LG-14/LG-8 stanzas above). Pinning
# LG-15's *message* would require either routing argparse's error output
# through something the harness captures (e.g. redirect_stderr, or
# making run_cli_argv catch SystemExit and return the message), which is
# a code change, not a test-only pin. Left open for design/look.md §8 to
# record; no `raises`-only stanza is added here because it would just
# duplicate LG-3/LG-14/LG-8 without pinning anything new about LG-15
# itself.

# --- LG-26: REMAINDER subcommands swallow unknown flags ---
# DESIRED: every subcommand should reject an unrecognized flag with an
# error and a non-zero exit, uniformly. `echo`, `wc`, and `git` are
# declared with nargs=argparse.REMAINDER and prefix_chars="\x00", so
# argparse never validates their arguments and a mistyped flag is
# silently swallowed as an ordinary operand instead.

# LG-26
call: ast_query.run_cli_str
arg cmdline = "echo --exclude foo"
expect return = "--exclude foo"

# LG-26
# wc's exact wording comes from the system wc binary and can vary across
# machines, so assert a stable fragment rather than an exact `expect
# return`.
call: ast_query.run_cli_str
arg cmdline = "wc -l libs/ast_query/fixtures/sample.py --exclude foo"
expect contains = "unrecognized option"

# LG-26: contrast only — cat and head already reject the same mistyped
# flag correctly (argparse SystemExit). This is the RIGHT behaviour;
# whoever closes LG-26 should make echo/wc/git converge to THIS, not
# "fix" these two by mistake.
call: ast_query.run_cli_str
arg cmdline = "cat libs/ast_query/fixtures/sample.py --exclude foo"
expect raises = "SystemExit"

# LG-26
call: ast_query.run_cli_str
arg cmdline = "head 2 libs/ast_query/fixtures/sample.py --exclude foo"
expect raises = "SystemExit"

# --- LG-12: the permitted git subcommand set is undiscoverable ---
# Rewritten 2026-08-23: the original row here judged LG-12 unpinnable (a
# discoverability observation about one worker's mistake, not a bug) and
# left only this comment. That judgement is overturned — a real defect
# was found. `_GIT_READONLY_SUBCOMMANDS` (ast_query.py:1142) holds 24
# entries and nothing prints them. `look git -h` is parsed as a

# the allowlist is the one gesture that gets rejected by it. You discover
# the boundary only by hitting it. This row is solely about the permitted
# set being unprintable; the secondary question of whether `branch`/
# `worktree` should be admitted is closed — see design/look.md §9.5.
# Full history: design/look.md §8.1 LG-12.

# LG-12
