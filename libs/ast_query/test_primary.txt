# --- run context: the two channels (design/look.md §9.1.1, §9.1.2) ---
# Each scenario below uses its own fresh ctx so stanzas do not interfere.

call: ast_query.new_run_context
save as: ctx_empty

call: ast_query.render_run_output
arg data = "hits"
arg ctx = $ctx_empty
expect return = "hits"
