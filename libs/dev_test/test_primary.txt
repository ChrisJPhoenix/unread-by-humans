# unmatched_filter_message: given a "libs/..." value, names the offending
# value and suggests the bare remainder ("orchestrator_plan").
call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[\"libs/orchestrator_plan\"]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect contains = "libs/orchestrator_plan"

call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[\"libs/orchestrator_plan\"]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect contains = "orchestrator_plan"

# unmatched_filter_message: a value with no "libs/" prefix names the value,
# and pins the exact full string so no bogus suggestion sneaks in — `expect
# contains` has no "does not contain" form, so this stanza uses `expect
# return =` against the exact text instead.
call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[\"nosuchlib\"]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect contains = "nosuchlib"

call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[\"nosuchlib\"]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect return = "no test plans matched libs=['nosuchlib'] under /repo/libs (mode=quick) — a filter names a library directory, not a path"

# unmatched_filter_message: a non-string entry in the filter list (e.g. an
# accidental int) is skipped rather than raising on .startswith — this is
# defect 1's shape: totality here is what keeps the SOLO branch from crashing
# on a malformed value instead of printing a message.
call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[\"libs/foo\", 3]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect return = "no test plans matched libs=['libs/foo', 3] under /repo/libs (mode=quick) — a filter names a library directory, not a path; try 'foo' instead"

# unmatched_filter_message: an empty filter list contributes no suggestions —
# pins totality at the other boundary (a non-list lib_filter is covered by
# filter_matched_no_plans's own None case above; this covers the empty-list
# shape of the same total contract).
call: dev_test.unmatched_filter_message_json
arg lib_filter_json = "[]"
arg libs_dir = "/repo/libs"
arg mode = "quick"
expect return = "no test plans matched libs=[] under /repo/libs (mode=quick) — a filter names a library directory, not a path"

# no_plans_found_message: the distinct "no filter, no plans at all" complaint

call: dev_test.no_plans_found_message

# next_timeout_seconds: the ceiling only ratchets up once a run consumes more
# than two thirds of it; exactly two thirds is not "more than" and leaves it
# unchanged, and the new ceiling is 2x the observed duration, not 2x the old
# ceiling.

# next_timeout_seconds: well under 2/3 of 900 (600) — unchanged.
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 100.0
arg current_timeout_seconds = 900
expect return = 900

# next_timeout_seconds: exactly 2/3 of 900 — not "more than", unchanged.
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 600.0
arg current_timeout_seconds = 900
expect return = 900

# next_timeout_seconds: just over the line — doubled off the observed duration.
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 601.0
arg current_timeout_seconds = 900
expect return = 1202

# next_timeout_seconds: fractional elapsed rounds up (ceiling, not truncation).
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 700.5
arg current_timeout_seconds = 900
expect return = 1401

# next_timeout_seconds: near the ceiling — doubled.
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 850.0
arg current_timeout_seconds = 900
expect return = 1700

# next_timeout_seconds: stability check — a second run of the same duration
# against the new ceiling must not ratchet again.
call: dev_test.next_timeout_seconds
arg elapsed_seconds = 601.0
arg current_timeout_seconds = 1202
expect return = 1202

# watchdog_timeout_message: names the elapsed time, the ceiling that tripped
# it, the mode that ceiling governs, and — the point of the phase argument —
# WHERE it hung. No running live test means the local domains, which is a real
# wedge to hunt rather than a ceiling to raise.
call: dev_test.watchdog_timeout_message
arg elapsed_seconds = 901.0
arg limit_seconds = 900.0
arg mode = "full"
arg running_live_test = ""
expect return = "dev test: no result after 901s, over the 900s full ceiling in libs/dev_test/timeout_seconds.txt — treating this as a hang. It timed out in the LOCAL test domains, with no live-API test running, so this is a real hang: find the wedged test rather than raising the ceiling. A timed-out run never raises the ceiling itself; raise that file's full value by hand only if the local suite is legitimately slower."

# watchdog_timeout_message: the same trip inside a gated live test is NOT a
# code hang — it is a real Claude session over the network — so the message
# names the test and says re-run.
call: dev_test.watchdog_timeout_message
arg elapsed_seconds = 145.0
arg limit_seconds = 145.0
arg mode = "quick"
arg running_live_test = "workshop_session_driver/live_test_secondlight_permissions"
expect return = "dev test: no result after 145s, over the 145s quick ceiling in libs/dev_test/timeout_seconds.txt — treating this as a hang. It timed out in the LIVE-API test workshop_session_driver/live_test_secondlight_permissions, which drives a real Claude session over the network — an occasional overrun there is expected and is not a code hang. Re-run it, or raise that file's quick value by hand; a timed-out run never raises it itself."

# parse_timeout_file_json failures: a bad ceiling file is fatal by design — a
# silent default would hide the fact that the ceiling in force is no longer
# the value anyone actually chose (see TimeoutFileUnusable).

# parse_timeout_file_json: empty text defines no modes.
call: dev_test.parse_timeout_file_json
arg text = ""
expect raises = "TimeoutFileUnusable"

# parse_timeout_file_json: only one of the two required modes present.
call: dev_test.parse_timeout_file_json
arg text = "quick 75\n"
expect raises = "TimeoutFileUnusable"

# parse_timeout_file_json: an unknown mode.
call: dev_test.parse_timeout_file_json
arg text = "quick 75\nfull 240\nsideways 9\n"
expect raises = "TimeoutFileUnusable"

# parse_timeout_file_json: a repeated mode.
call: dev_test.parse_timeout_file_json
arg text = "quick 75\nquick 80\nfull 240\n"
expect raises = "TimeoutFileUnusable"

# parse_timeout_file_json: a non-integer value.

expect raises = "TimeoutFileUnusable"

# parse_timeout_file_json: a zero value.
call: dev_test.parse_timeout_file_json

# render_timeout_file: renders {mode: seconds} back to the file's two-line
# text in CEILING_MODES order — the inverse of parse_timeout_file. Passed a
# real dict via save-as/$var chaining (the plan format has no dict-literal
# syntax to spell one directly).
call: dev_test.parse_timeout_file
arg text = "quick 75\nfull 240\n"
save as: ceilings

call: dev_test.render_timeout_file
arg ceilings = $ceilings
expect return = "quick 75\nfull 240\n"
