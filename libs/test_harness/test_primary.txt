# test_harness: self-test
#
# The harness exercises itself by running inline plans through
# `run_text` and inspecting the resulting PlanResult. Each block
# pins down one behaviour.

# --- expect contains: substring check against the return value -----------

# PASS: the return value contains the expected substring.
call: test_harness.run_text
arg text = "call: demo.identity\narg x = \"hello world\"\nexpect contains = \"world\""
save as: contains_pass

call: test_harness.result_status
arg result = $contains_pass
expect return = "pass"

# FAIL: the return value does not contain the expected substring.
call: test_harness.run_text
arg text = "call: demo.identity\narg x = \"hello world\"\nexpect contains = \"missing\""
save as: contains_fail

call: test_harness.result_status
arg result = $contains_fail
expect return = "fail"

# --- Parsing: stanza counting -----------------------------------------

# Single stanza.
call: test_harness.count_stanzas
arg text = "call: demo.identity\narg x = 1\nexpect return = 1"
expect return = 1

# Two stanzas separated by blank line.
call: test_harness.count_stanzas
arg text = "call: demo.identity\narg x = 1\n\ncall: demo.identity\narg x = 2"
expect return = 2

# Comments only -> zero stanzas.
call: test_harness.count_stanzas
arg text = "# just a comment\n\n# another"
expect return = 0

# --- Plan execution: success path -------------------------------------

call: test_harness.run_text
arg text = "call: demo.add\narg a = 2\narg b = 3\nexpect return = 5"
save as: passing_result

call: test_harness.result_status
arg result = $passing_result
expect return = "pass"

call: test_harness.result_step_count
arg result = $passing_result
expect return = 1

# --- Mismatch detection -----------------------------------------------

call: test_harness.run_text
arg text = "call: demo.add\narg a = 2\narg b = 3\nexpect return = 99"
save as: failing_result

call: test_harness.result_status
arg result = $failing_result
expect return = "fail"

# --- raises ------------------------------------------------------------

call: test_harness.run_text
arg text = "call: demo.always_raises\nexpect raises = \"ValueError\""
save as: raises_result

call: test_harness.result_status
arg result = $raises_result
expect return = "pass"

# An unhandled exception is reported as an error.
call: test_harness.run_text
arg text = "call: demo.always_raises"
save as: unhandled_error

call: test_harness.result_status
arg result = $unhandled_error
expect return = "error"

# --- Parse error reported as an error result, not a crash -------------

call: test_harness.run_text
arg text = "not a call line"
save as: bad_parse

call: test_harness.result_status
arg result = $bad_parse
expect return = "error"

# --- Stub-boundary handling -------------------------------------------

# Calling a @stub from a non-stub-own plan halts with status=stubbed.
call: test_harness.run_text
arg text = "call: demo.future_thing\nexpect return = 0"
save as: stubbed_result

call: test_harness.result_status
arg result = $stubbed_result
expect return = "pass"

call: test_harness.result_stubbed_calls
arg result = $stubbed_result
save as: stubbed_calls_list

# We can't directly compare list contents to a literal yet, so
# we count the length via a tiny helper expectation: stubbed
# plans terminate after exactly one step.
call: test_harness.result_step_count
arg result = $stubbed_result
expect return = 1

# --- Discovery --------------------------------------------------------

# discover_plans returns at least one path for the libs/ tree.
# We don't pin an exact count here (it would brittlely couple
# the self-test to the number of libraries in the project); we
# just check the call doesn't raise.
call: test_harness.discover_plans
arg libs_dir = "libs"
arg mode = "quick"

# plan_glob_for_mode: quick takes every test_primary*.txt, so a library
# whose plan is split per feature keeps its whole quick-tier coverage by
# naming each piece test_primary_<feature>.txt.
call: test_harness.plan_glob_for_mode
arg mode = "quick"
expect return = "test_primary*.txt"

# plan_glob_for_mode: full additionally picks up scenario and stub plans.
call: test_harness.plan_glob_for_mode
arg mode = "full"
expect return = "test_*.txt"

# --- Stub coverage check ----------------------------------------------

call: test_harness.check_stub_coverage
arg libs_dir = "libs"
save as: coverage

call: test_harness.coverage_ok
arg report = $coverage
expect return = true

# --- Learn mode --------------------------------------------------------

# When the harness reruns a plan whose expects already match the
# actuals, learn-mode is a no-op. We can verify by running the
# same passing inline twice and confirming both pass.
call: test_harness.run_text
arg text = "call: demo.identity\narg x = 7\nexpect return = 7"
save as: learn_baseline

call: test_harness.result_status
arg result = $learn_baseline
expect return = "pass"

# --- Normalization (normalize:) ---------------------------------------

# PASS: a normalize rule masks the varying number so the masked expect matches.
call: test_harness.run_text
arg text = "call: demo.labeled_count\nnormalize: count: \\d+ => count: <n>\nexpect return = \"count: <n>\""
save as: normalize_pass

call: test_harness.result_status
arg result = $normalize_pass
expect return = "pass"

# FAIL: without the normalize rule, "count: 42" != the literal "count: <n>".
call: test_harness.run_text
arg text = "call: demo.labeled_count\nexpect return = \"count: <n>\""
save as: normalize_absent

call: test_harness.result_status
arg result = $normalize_absent
expect return = "fail"

# LEARN round-trip: learn rewrites the wrong expect to the MASKED actual,
# and the learned plan then re-passes (proves learn writes the masked form).
call: test_harness.learned_text
arg text = "call: demo.labeled_count\nnormalize: count: \\d+ => count: <n>\nexpect return = \"wrong\""
save as: learned_plan

call: test_harness.run_text
arg text = $learned_plan
save as: relearned_result

call: test_harness.result_status
arg result = $relearned_result
expect return = "pass"

# --- Float tolerance: only float-spelled targets get a tolerance band ----

# Same call, two spellings. `demo.add(1, 0.0000001)` returns 1.0000001.
# Target spelled as a float (1.0) accepts the 1e-7 gap (default 1e-6 band).
call: test_harness.run_text
arg text = "call: demo.add\narg a = 1\narg b = 0.0000001\nexpect return = 1.0"
save as: float_spelled_pass

call: test_harness.result_status
arg result = $float_spelled_pass
expect return = "pass"

# Target spelled as an int (1) demands exact equality, so the gap fails.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 1\narg b = 0.0000001\nexpect return = 1"
save as: int_spelled_fail

call: test_harness.result_status
arg result = $int_spelled_fail
expect return = "fail"

# Default 1e-6 band absorbs float rounding: 0.1 + 0.2 == 0.30000000000000004.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.1\narg b = 0.2\nexpect return = 0.3"
save as: default_band_pass

call: test_harness.result_status
arg result = $default_band_pass
expect return = "pass"

# Per-stanza `tolerance: 0` overrides the band, forcing exact equality.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.1\narg b = 0.2\ntolerance: 0\nexpect return = 0.3"
save as: zero_tol_fail

call: test_harness.result_status
arg result = $zero_tol_fail
expect return = "fail"

# Per-ELEMENT spelling inside a block. The returned row is (2.0000001, 3.0).
# Element 0 spelled as int (2) -> exact -> the 1e-7 gap fails the whole block.
call: test_harness.run_text
arg text = "call: demo.array_just_above_two_three\nexpect return =\n  2 3.0"
save as: block_int_elem_fail

call: test_harness.result_status
arg result = $block_int_elem_fail
expect return = "fail"

# Element 0 spelled as float (2.0) -> tolerant -> both elements pass.
call: test_harness.run_text
arg text = "call: demo.array_just_above_two_three\nexpect return =\n  2.0 3.0"
save as: block_float_elem_pass

call: test_harness.result_status
arg result = $block_float_elem_pass
expect return = "pass"

# Zero target: a relative band would be 1e-6 * 0 == 0, so a float-spelled
# 0.0 (one decimal place) instead gets the band 1e-6 * 10**(1-1) = 1e-6. 1e-7 ~ 0.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.0000001\narg b = 0.0\nexpect return = 0.0"
save as: zero_target_floor_pass

call: test_harness.result_status
arg result = $zero_target_floor_pass
expect return = "pass"

# That 0.0 band is 1e-6, so a 1e-3 gap still fails.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.001\narg b = 0.0\nexpect return = 0.0"
save as: zero_target_floor_fail

call: test_harness.result_status
arg result = $zero_target_floor_fail
expect return = "fail"

# --- Zero target: written decimal places tighten the band 10x each --------

# demo.add(0.0000005, 0.0) returns 5e-7. Against 0.0 the band is 1e-6 -> pass;
# against 0.00 the band tightens to 1e-7 -> the same 5e-7 gap now fails.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.0000005\narg b = 0.0\nexpect return = 0.0"
save as: zero_1dp_pass

call: test_harness.result_status
arg result = $zero_1dp_pass
expect return = "pass"

call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.0000005\narg b = 0.0\nexpect return = 0.00"
save as: zero_2dp_fail

call: test_harness.result_status
arg result = $zero_2dp_fail
expect return = "fail"

# demo.add(0.00000005, 0.0) returns 5e-8. Against 0.00 the band is 1e-7 -> pass;
# against 0.000 the band tightens to 1e-8 -> the same 5e-8 gap now fails.
call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.00000005\narg b = 0.0\nexpect return = 0.00"
save as: zero_2dp_pass

call: test_harness.result_status
arg result = $zero_2dp_pass
expect return = "pass"

call: test_harness.run_text
arg text = "call: demo.add\narg a = 0.00000005\narg b = 0.0\nexpect return = 0.000"
save as: zero_3dp_fail

call: test_harness.result_status
arg result = $zero_3dp_fail
expect return = "fail"

# --- \xHH escapes: the grammar's general control-byte form ---------------

# Round-trip in both directions at once. The inner plan writes \x0D, so if the
# escape works `demo.identity` receives a real CR and learn rewrites the wrong
# expect to the CR's own spelling, \x0d. If it did NOT work the argument would
# be the five characters "ax0Db" and the learned line would say so.
call: test_harness.learned_text
arg text = "call: demo.identity\narg x = \"a\\x0Db\"\nexpect return = \"wrong\""
expect contains = "expect return = \"a\\x0db\""

# \x with digits that are not hex is a parse error, not a silent fallback.
call: test_harness.run_text
arg text = "call: demo.identity\narg x = \"a\\xZZb\"\nexpect return = 1"
save as: bad_hex_escape

call: test_harness.result_status
arg result = $bad_hex_escape
expect return = "error"

# \x at the very end of a string, with no digits at all, is the same error.
call: test_harness.run_text
arg text = "call: demo.identity\narg x = \"trailing\\x\"\nexpect return = 1"
save as: truncated_hex_escape

call: test_harness.result_status
arg result = $truncated_hex_escape
expect return = "error"
